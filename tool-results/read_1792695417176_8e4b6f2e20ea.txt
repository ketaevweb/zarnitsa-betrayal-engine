   164→/* ---------------- модалка события / разговора ---------------- */

   166→/** печатающийся текст: ~0.9 с на полный текст, клик — показать сразу.
   167→ *  Компонент монтируется с key={event.id} — сброс состояния бесплатен. */
   168→function useTypedText(text: string[]) {
   169→  const full = useMemo(() => text.join(""), [text]);
   170→  const [shown, setShown] = useState(0);
   171→  useEffect(() => {
   172→    if (!full) return;
   173→    const step = Math.max(4, Math.round(full.length / 55));
   174→    const iv = setInterval(() => {
   175→      setShown((s) => Math.min(full.length, s + step));
   176→    }, 16);
   177→    return () => clearInterval(iv);
   178→  }, [full]);
   179→  return { full, shown, done: shown >= full.length, complete: () => setShown(full.length) };
   180→}

   182→function EventModal() {
   183→  const ev = useZarnitsa((s) => s.activeEvent);
   184→  const choose = useZarnitsa((s) => s.choose);
   185→  const typed = useTypedText(ev?.text ?? [""]);
   186→
   187→  useEffect(() => {
   188→    // разговоры открываются мягче, чем ключевые события (у тех — «удар» в openEvent)
   189→    if (ev?.id.startsWith("talk_")) sfxOpen();
   190→  }, [ev]);
   191→
   192→  if (!ev) return null;
   193→  const isTalk = ev.id.startsWith("talk_");
   194→
   195→  // распределение напечатанных символов по абзацам (без мутаций и без хуков)
   196→  const starts: number[] = [];
   197→  let sum = 0;
   198→  for (const p of ev.text) {
   199→    starts.push(sum);
   200→    sum += p.length;
   201→  }
   202→  const parts = ev.text.map((p, i) => {
   203→    const start = starts[i];
   204→    const take = typed.shown > start ? Math.min(p.length, typed.shown - start) : 0;
   205→    return { p, take };
   206→  });
   207→  const typingIdx = parts.findIndex((x) => x.take < x.p.length);
   208→
   209→  return (
   210→    <div className="pointer-events-auto absolute inset-0 z-30 flex items-center justify-center bg-black/55 p-4">
   211→      <section
   212→        className="z-panel z-fadein max-h-[86vh] w-full max-w-2xl overflow-y-auto p-5 sm:p-7"
   213→        aria-label={ev.title}
   214→        onClick={typed.complete}
   215→      >
   216→        <div className="flex flex-wrap items-center gap-2">
   217→          <span className="rounded-sm border border-[#32233a] px-2 py-0.5 text-[11px] text-[#9a8f96]">
   218→            {isTalk ? "разговор у костра" : `Ночь ${ev.day} · ${ev.phase}`}
   219→          </span>
   220→          {ev.isOrder && (
   221→            <span className="rounded-sm border border-[#b3542f] bg-[#b3542f]/10 px-2 py-0.5 text-[11px] font-semibold text-[#b3542f]">
   222→              ПАУЗА ПРИСЯГИ · приказ
   223→            </span>
   224→          )}
   225→          {ev.title.startsWith("⸸") && (
   226→            <span className="rounded-sm border border-[#c9a227] bg-[#c9a227]/10 px-2 py-0.5 text-[11px] font-semibold text-[#c9a227]">
   227→              катализатор
   228→            </span>
   229→          )}
   230→        </div>
   231→        <h2 className="z-heading mt-2 text-2xl font-semibold sm:text-3xl">
   232→          {ev.title.replace("⸸ ", "")}
   233→        </h2>
   234→        <div className="mt-3 space-y-2.5">
   235→          {parts.map(({ p, take }, i) => (
   236→            <p key={i} className="text-sm leading-relaxed text-[#d6cfc2] sm:text-[15px]">
   237→              {p.slice(0, take)}
   238→              {i === typingIdx && (
   239→                <span className="text-[#c9a227]">▍</span>
   240→              )}
   241→            </p>
   242→          ))}
   243→        </div>
   244→        {!typed.done && (
   245→          <p className="mt-1.5 text-right text-[10px] uppercase tracking-[0.2em] text-[#9a8f96]/70">
   246→            клик — дальше
   247→          </p>
   248→        )}
   249→        <div className="mt-5 space-y-2">
   250→          {ev.choices.map((c, i) => (
   251→            <button
   252→              key={c.id}
   253→              onClick={(e) => {
   254→                e.stopPropagation();
   255→                choose(i);
   256→              }}
   257→              className="z-choice"
   258→            >
   259→              <span className="block text-sm font-medium sm:text-[15px]">{c.label}</span>
   260→              {c.note && (
   261→                <span className="mt-0.5 block text-xs italic text-[#9a8f96]">{c.note}</span>
   262→              )}
   263→            </button>
   264→          ))}
   265→        </div>
   266→      </section>
   267→    </div>
   268→  );
   269→}

   614→/* ---------------- виньетка напряжения (M0.7) ---------------- */
   615→
   616→function Vignette() {
   617→  const party = useZarnitsa((s) => s.state.party);
   618→  const screen = useZarnitsa((s) => s.screen);
   619→
   620→  let maxLevel = 0;
   621→  let frozen = 0;
   622→  for (const cs of Object.values(party)) {
   623→    if (cs.gone) continue;
   624→    if (cs.level > maxLevel) maxLevel = cs.level;
   625→    if (cs.frozen) frozen++;
   626→  }
   627→  const intensity = Math.min(1, (maxLevel + frozen * 0.5) / 5.5);
   628→  const hot = maxLevel >= 4 || frozen > 0;
   629→  const level = Math.round(intensity * 10) / 10;
   630→
   631→  useEffect(() => {
   632→    setHeartbeat(hot && screen === "play" ? 0.3 + level * 0.6 : 0);
   633→    return () => setHeartbeat(0);
   634→  }, [hot, screen, level]);
   635→
   636→  if (screen === "title" || screen === "end" || intensity < 0.02) return null;
   637→  return (
   638→    <>
   639→      <div
   640→        className="pointer-events-none absolute inset-0 z-[15]"
   641→        style={{
   642→          background: `radial-gradient(ellipse at center, transparent 40%, rgba(90,16,32,${(
   643→            0.55 * intensity
   644→          ).toFixed(3)}) 100%)`,
   645→        }}
   646→      />
   647→      {hot && screen === "play" && (
   648→        <div
   649→          className="z-heartbeat pointer-events-none absolute inset-0 z-[15]"
   650→          style={{
   651→            background:
   652→              "radial-gradient(ellipse at center, transparent 52%, rgba(140,31,40,0.42) 100%)",
   653→          }}
   654→        />
   655→      )}
   656→    </>
   657→  );
   658→}

   702→      <PartyDrawer />
